cmake_minimum_required(VERSION 3.24)
project(AIInferenceGateway VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(GATEWAY_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
if(GATEWAY_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  add_link_options(-fsanitize=address,undefined -no-pie)
endif()
add_library(llhttp STATIC third_party/llhttp/src/api.c third_party/llhttp/src/http.c third_party/llhttp/src/llhttp.c)
target_include_directories(llhttp PUBLIC third_party/llhttp/include)
add_library(gateway_foundation STATIC src/base/Buffer.cpp src/net/Socket.cpp src/timer/TimerQueue.cpp src/http/HttpParser.cpp src/http/StreamOutput.cpp src/http/HttpNormalization.cpp src/gateway/TextRequest.cpp)
target_include_directories(gateway_foundation PUBLIC src)
target_include_directories(gateway_foundation SYSTEM PRIVATE third_party)
target_link_libraries(gateway_foundation PUBLIC llhttp)
target_compile_options(gateway_foundation PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_library(gateway_runtime STATIC src/admission/AdmissionController.cpp src/routing/ModelRouter.cpp src/gateway/BufferBudget.cpp src/upstream/ConnectionPool.cpp src/net/EventLoop.cpp src/net/Channel.cpp src/net/Epoller.cpp src/app/GatewayConfig.cpp src/app/SignalSource.cpp src/gateway/SessionIo.cpp src/gateway/ProxySession.cpp src/gateway/GatewayRuntime.cpp)
target_link_libraries(gateway_runtime PUBLIC gateway_foundation)
target_include_directories(gateway_runtime SYSTEM PRIVATE third_party)
target_compile_options(gateway_runtime PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(ai-inference-gateway src/app/main.cpp)
target_link_libraries(ai-inference-gateway PRIVATE gateway_runtime)
include(CTest)
