cmake_minimum_required(VERSION 3.24)
project(AIInferenceGateway VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(GATEWAY_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
if(GATEWAY_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  add_link_options(-fsanitize=address,undefined -no-pie)
endif()
add_library(llhttp STATIC third_party/llhttp/src/api.c third_party/llhttp/src/http.c third_party/llhttp/src/llhttp.c)
target_include_directories(llhttp PUBLIC third_party/llhttp/include)
add_library(gateway_foundation STATIC src/base/Buffer.cpp src/net/Socket.cpp src/timer/TimerQueue.cpp src/http/HttpParser.cpp src/http/HttpNormalization.cpp src/gateway/TextRequest.cpp)
target_include_directories(gateway_foundation PUBLIC src)
target_include_directories(gateway_foundation SYSTEM PRIVATE third_party)
target_link_libraries(gateway_foundation PUBLIC llhttp)
target_compile_options(gateway_foundation PRIVATE -Wall -Wextra -Wpedantic -Werror)
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_executable(FoundationTests tests/FoundationTests.cpp)
  target_link_libraries(FoundationTests PRIVATE gateway_foundation)
  add_executable(ProtocolTests tests/ProtocolTests.cpp)
  target_link_libraries(ProtocolTests PRIVATE gateway_foundation)
  add_executable(ParseResponse tests/ParseResponse.cpp)
  target_link_libraries(ParseResponse PRIVATE gateway_foundation)
  add_test(NAME Foundation COMMAND FoundationTests)
  add_test(NAME Protocol COMMAND ProtocolTests)
  add_test(NAME AssertionFailure COMMAND FoundationTests --intentional-failure)
  set_tests_properties(AssertionFailure PROPERTIES WILL_FAIL TRUE)
  add_test(NAME MockBackend COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/MockBackend_test.py $<TARGET_FILE:ParseResponse>)
  add_test(NAME Manifest COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/Manifest_test.py)
  set_tests_properties(Foundation Protocol AssertionFailure MockBackend Manifest PROPERTIES TIMEOUT 120)
endif()
